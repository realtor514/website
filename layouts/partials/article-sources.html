{{- /* Sources officielles d un article, lues dans meta/fr/.json (monte en site.Data.sources). Les quatre langues partagent la fiche de la version francaise: on la retrouve par la traduction francaise. Usage: {{ $src := partial "article-sources.html" . }} Renvoie une liste de {url, title, publisher, consulted}, vide si l article n a pas de fiche. Le champ interne `supports` n est jamais renvoye: ce sont des notes de travail, pas du contenu publie. */ -}} {{- $fr := cond (eq .Lang "fr") . nil -}} {{- if not $fr -}} {{- range .Translations -}}{{- if eq .Lang "fr" -}}{{- $fr = . -}}{{- end -}}{{- end -}} {{- end -}} {{- $out := slice -}} {{- $seen := slice -}} {{- with $fr -}} {{- with .File -}} {{- with index site.Data.sources .ContentBaseName -}} {{- range .sources -}} {{- if and .url .title (hasPrefix .url "http") (not (in $seen .url)) -}} {{- $seen = $seen | append .url -}} {{- $out = $out | append (dict "url" .url "title" (replaceRE "[–—―]" "-" .title) "publisher" (replaceRE "[–—―]" "-" (.publisher | default "")) "consulted" (.consulted | default "")) -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- return $out -}}
